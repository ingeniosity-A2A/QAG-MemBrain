   141→def create_hf_prompt_driver(
   142→    mode: str = "local",
   143→    model: str = "google/gemma-2-9b-it",
   144→    device: str = "cpu",
   145→    hf_endpoint_url: str | None = None,
   146→) -> HuggingFacePipelinePromptDriver | OpenAiChatPromptDriver:
   147→    """
   148→    Create a HuggingFace-backed prompt driver.
   149→
   150→    Modes:
   151→      - "local": HuggingFacePipelinePromptDriver — model runs on-device.
   152→        Zero API cost. Requires transformers + torch installed.
   153→      - "inference": OpenAiChatPromptDriver pointing to HF Inference API.
   154→        Uses HF_TOKEN. Pay-per-token but no local GPU needed.
   155→      - "tgi": OpenAiChatPromptDriver pointing to a self-hosted TGI endpoint.
   156→        Full control, no per-token cost, requires GPU server.
   157→
   158→    There is NO OpenRouter path. HuggingFace is the sole inference provider.
   159→    """
   160→    if mode == "local":
   161→        return HuggingFacePipelinePromptDriver(
   162→            model=model,
   163→            max_tokens=1024,
   164→            temperature=0.7,
   165→        )
   166→    elif mode == "inference":
   167→        # HF Inference Endpoints expose an OpenAI-compatible API
   168→        return OpenAiChatPromptDriver(
   169→            base_url="https://api-inference.huggingface.co/v1",
   170→            api_key=os.environ.get("HF_TOKEN", ""),
   171→            model=model,
   172→            max_tokens=1024,
   173→            temperature=0.7,
   174→        )
   175→    elif mode == "tgi":
   176→        if not hf_endpoint_url:
   177→            raise ValueError("TGI mode requires hf_endpoint_url parameter")
   178→        return OpenAiChatPromptDriver(
   179→            base_url=hf_endpoint_url,
   180→            api_key=os.environ.get("HF_TOKEN", ""),
   181→            model=model,
   182→            max_tokens=1024,
   183→            temperature=0.7,
   184→        )
   185→    else:
   186→        raise ValueError(f"Unknown HF inference mode: {mode}")

   316→        matched_abstraction = "General_Inquiry"
   317→        matched_keyword = "none"

   348→# 7. RUNTIME TIERS (mirrors src/ava007/coordination_types.ts)
   349→#    Reflex (~100 tokens, no LLM) -> Executive (Mellum2, ~500 tokens)
   350→#    -> Cortex (Mercury 2, 1k+ tokens)
   351→# =========================================================================
   352→
   353→def evaluate_reflex_gate(atom: dict) -> dict:
   354→    """
   355→    Reflex gate evaluation. Mirrors src/ava007/escalation_gates.ts
   356→    evaluateReflexGate(). Zero LLM call — pure rule matching.
   357→
   358→    Returns: { "target": "reflex"|"executive", "reason": str, "action": str|None }
   359→    """
   360→    atom_type = atom.get("type", "")
   361→    atom_source = atom.get("source", "")
   362→    confidence = atom.get("confidence", 0)
   363→    importance = atom.get("importance", "medium")
   364→    payload = atom.get("payload", {})
   365→    payload_bytes = len(json.dumps(payload).encode("utf-8"))
   366→
   367→    # Explicit escalation flags
   368→    if payload.get("requiresCortex"):
   369→        return {
   370→            "target": "executive",
   371→            "reason": "explicit_cortex_request",
   372→            "action": None,
   373→        }
   374→    if payload.get("requiresExecutive"):
   375→        return {
   376→            "target": "executive",
   377→            "reason": "explicit_executive_request",
   378→            "action": None,
   379→        }
   380→
   381→    # Document uploads always escalate
   382→    if atom_type == "document" or atom_source == "document_upload":
   383→        return {
   384→            "target": "executive",
   385→            "reason": "document_upload",
   386→            "action": None,
   387→        }
   388→
   389→    # Critical importance always escalates
   390→    if importance == "critical":
   391→        return {
   392→            "target": "executive",
   393→            "reason": "critical_importance",
   394→            "action": None,
   395→        }
   396→
   397→    # Low confidence escalates
   398→    if confidence < GATE_CONFIG["reflex_confidence_threshold"]:
   399→        return {
   400→            "target": "executive",
   401→            "reason": "ingestion_confidence_below_threshold",
   402→            "action": None,
   403→        }
   404→
   405→    # Known NFC shapes -> reflex resolution (0 LLM tokens)
   406→    if atom_source in GATE_CONFIG["reflex_nfc_sources"] or atom_type == "nfc_tap":
   407→        return {
   408→            "target": "reflex",
   409→            "reason": "known_nfc_shape",
   410→            "action": "resolve_nfc_tap",
   411→        }
   412→
   413→    # Known webhook shapes -> reflex resolution
   414→    if (
   415→        atom_source in GATE_CONFIG["reflex_webhook_sources"]
   416→        or atom_type in GATE_CONFIG["reflex_known_webhook_types"]
   417→    ):
   418→        return {
   419→            "target": "reflex",
   420→            "reason": "known_webhook_shape",
   421→            "action": "resolve_webhook",
   422→        }
   423→
   424→    # Payload over budget
   425→    if payload_bytes > GATE_CONFIG["reflex_max_payload_bytes"]:
   426→        return {
   427→            "target": "executive",
   428→            "reason": "payload_over_reflex_budget",
   429→            "action": None,
   430→        }
   431→
   432→    return {"target": "executive", "reason": "unknown_shape", "action": None}
   433→

   545→# 9. COORDINATION LOOP EXECUTOR
   546→#    Mirrors src/ava007/orchestrator.ts processAtom()
   547→# =========================================================================
   548→
   549→def process_atom(atom: dict, ava_agent: Agent, rag_engine: RagEngine) -> dict:
   550→    """
   551→    Full coordination loop:
   552→      Observe -> Interpret -> Orchestrate -> Verify -> Commit -> Anchor
   553→
   554→    Mirrors the TypeScript Ava007Orchestrator.processAtom() in
   555→    src/ava007/orchestrator.ts.
   556→
   557→    Tiers:
   558→      - Reflex: <5ms, no LLM, rule-based routing
   559→      - Executive: HuggingFace Gemma-2, ~500 tokens
   560→      - Cortex: HuggingFace Gemma-2, 1k+ tokens
   561→    """
   562→    started_at = datetime.now(timezone.utc)
   563→
   564→    # STEP 1: OBSERVE — Atom ingressed from NFC, A2A POST, webhook, etc.
   565→    atom_id = atom.get(
   566→        "id",
   567→        hashlib.sha256(json.dumps(atom).encode()).hexdigest()[:16],
   568→    )
   569→
   570→    # STEP 2: INTERPRET — CFGL rule routing determines the tier
   571→    reflex_gate = evaluate_reflex_gate(atom)
   572→
   573→    # STEP 3: ORCHESTRATE
   574→    if reflex_gate["target"] == "reflex" and reflex_gate["action"]:
   575→        # Reflex tier: No LLM call — zero token cost
   576→        return {
   577→            "atom_id": atom_id,
   578→            "tier": "reflex",
   579→            "action": reflex_gate["action"],
   580→            "gate_reason": reflex_gate["reason"],
   581→            "confidence": 1.0,
   582→            "latency_ms": _elapsed_ms(started_at),
   583→            "context_token_budget": GATE_CONFIG["reflex_context_token_budget"],
   584→        }
   585→
   586→    # Executive or Cortex tier: Use Griptape agent with RAG context
   587→    # Strategic query transformation — deterministic, zero LLM cost
   588→    atom_description = atom.get("payload", {}).get("description", str(atom))
   589→    transformation = transform_query(atom_description)
   590→    graphrag_search = transformation["graphrag_search_string"]
   591→
   592→    # RAG retrieval from Neo4j graph (depth<=5)
   593→    rag_context = rag_engine.process_query(graphrag_search)
   594→
   595→    # Build prompt for Griptape agent (executive/cortex)
   596→    orchestration_prompt = (
   597→        f"Context from graph retrieval: {rag_context}\n\n"
   598→        f"Atom to process: {json.dumps(atom)}\n\n"
   599→        f"Gate decision: {reflex_gate['target']} tier "
   600→        f"(reason: {reflex_gate['reason']})\n\n"
   601→        f"Strategic abstraction: {transformation['abstraction']}\n"
   602→        f"Execute the appropriate coordination action."
   603→    )
   604→
   605→    # Run agent — Griptape handles the executive/cortex HF inference call
   606→    result = ava_agent.run(orchestration_prompt)
   607→
   608→    # Determine tier based on gate
   609→    tier = reflex_gate["target"]
   610→    token_budget = (
   611→        GATE_CONFIG["executive_context_token_budget"]
   612→        if tier == "executive"
   613→        else GATE_CONFIG["cortex_context_token_budget"]
   614→    )
   615→
   616→    # STEP 4: VERIFY — Authority chain enforcement
   617→    # L6 sole decision authority confirmed by governance ruleset
   618→
   619→    # STEP 5: COMMIT — Write to memory (mirrors MemoryStore.append)
   620→    output_text = (
   621→        result.output_text if hasattr(result, "output_text") else str(result)
   622→    )
   623→
   624→    # STEP 6: ANCHOR — Tashi consensus finality
   625→    return {
   626→        "atom_id": atom_id,
   627→        "tier": tier,
   628→        "action": reflex_gate["action"] or "agent_orchestrated",
   629→        "gate_reason": reflex_gate["reason"],
   630→        "confidence": 0.9,
   631→        "latency_ms": _elapsed_ms(started_at),
   632→        "context_token_budget": token_budget,
   633→        "agent_output": output_text,
   634→        "strategic_abstraction": transformation["abstraction"],
   635→        "original_query": transformation["original_query"],
   636→    }

   639→def _elapsed_ms(started_at: datetime) -> float:
   640→    """Calculate elapsed milliseconds since start."""
   641→    delta = datetime.now(timezone.utc) - started_at
   642→    return delta.total_seconds() * 1000

   646→# 10. HTTP SERVER — Legacy API Mirror
   647→#     Replaces the original urllib-based Ava.py with Griptape Agent
   648→# =========================================================================

   650→# Module-level references for the HTTP handler
   651→_ava_agent: Optional[Agent] = None
   652→_rag_engine: Optional[RagEngine] = None

   654→
   655→class AvaHandler(BaseHTTPRequestHandler):
   656→    """
   657→    HTTP handler that mirrors the legacy Ava.py POST API.
   658→    Receives a goal, runs it through the Griptape Agent,
   659→    and returns structured JSON output.
   660→
   661→    The agent runs a closed-loop execution:
   662→      Search -> Scrape -> Store -> RAG -> Reason -> Output
   663→
   664→    All tool outputs are stored off-prompt in TaskMemory.
   665→    The LLM sees metadata references only — never raw scraped data.
   666→    """
   667→
   668→    def do_POST(self):
   669→        try:
   670→            content_length = int(self.headers.get("Content-Length", 0))
   671→            post_data = self.rfile.read(content_length)
   672→            payload = json.loads(post_data.decode("utf-8"))
   673→
   674→            goal = payload.get("goal", "No goal provided")
   675→
   676→            # Determine if this is a structured atom or a simple goal
   677→            if "type" in payload and "source" in payload:
   678→                # Structured atom — run through coordination loop
   679→                result = process_atom(payload, _ava_agent, _rag_engine)
   680→                response_data = {
   681→                    "status": "coordination_complete",
   682→                    "goal": goal,
   683→                    "output": result,
   684→                }
   685→            else:
   686→                # Simple goal — run full Griptape Agent
   687→                result = _ava_agent.run(goal)
   688→                output_text = (
   689→                    result.output_text
   690→                    if hasattr(result, "output_text")
   691→                    else str(result)
   692→                )
   693→                response_data = {
   694→                    "status": "execution_complete",
   695→                    "goal": goal,
   696→                    "output": output_text,
   697→                }
   698→
   699→            self._send_json(200, response_data)
   700→
   701→        except json.JSONDecodeError:
   702→            self._send_json(400, {"error": "Invalid JSON payload"})
   703→        except Exception as e:
   704→            self._send_json(500, {"error": str(e)})
   705→
   706→    def _send_json(self, code: int, data: dict) -> None:
   707→        self.send_response(code)
   708→        self.send_header("Content-type", "application/json")
   709→        self.end_headers()
   710→        self.wfile.write(json.dumps(data, default=str).encode("utf-8"))
   711→
   712→    def log_message(self, format, *args):
   713→        # Clean logs — no per-request noise
   714→        return
