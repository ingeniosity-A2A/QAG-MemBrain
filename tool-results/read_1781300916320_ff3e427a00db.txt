   138→# 2. HUGGINGFACE INFERENCE CONFIGURATION
   139→#    Four deployment modes: hub (GPU-less), local pipeline, HF Inference API, HF TGI

   142→def create_hf_prompt_driver(
   143→    mode: str = "hub",
   144→    model: str = "mistralai/Mistral-7B-Instruct-v0.2",
   145→    device: str = "cpu",
   146→    hf_endpoint_url: str | None = None,
   147→) -> HuggingFaceHubPromptDriver | HuggingFacePipelinePromptDriver | OpenAiChatPromptDriver:
   148→    """
   149→    Create a HuggingFace-backed prompt driver.
   150→
   151→    Modes:
   152→      - "hub" (DEFAULT): HuggingFaceHubPromptDriver — GPU-less cloud inference.
   153→        Uses HF Inference API via Griptape's native Hub driver.
   154→        Zero local VRAM. Zero local compute. Requires HF_TOKEN.
   155→        Best for: production, edge devices, Termux, S25 Ultra, any GPU-less deployment.
   156→
   157→      - "local": HuggingFacePipelinePromptDriver — model runs on-device.
   158→        Zero API cost. Requires transformers + torch installed.
   159→        Best for: air-gapped environments, development with local GPU.
   160→
   161→      - "inference": OpenAiChatPromptDriver pointing to HF Inference API.
   162→        Uses OpenAI-compatible endpoint. Same cloud as "hub" but via different protocol.
   163→        Best for: compatibility with OpenAI tooling, streaming support.
   164→
   165→      - "tgi": OpenAiChatPromptDriver pointing to a self-hosted TGI endpoint.
   166→        Full control, no per-token cost, requires GPU server.
   167→
   168→    There is NO OpenRouter path. HuggingFace is the sole inference provider.
   169→    """
   170→    if mode == "hub":
   171→        return HuggingFaceHubPromptDriver(
   172→            model=model,
   173→            api_token=os.environ.get("HF_TOKEN", ""),
   174→            max_tokens=1024,
   175→            temperature=0.7,
   176→        )
   177→    elif mode == "local":
   178→        return HuggingFacePipelinePromptDriver(
   179→            model=model,
   180→            max_tokens=1024,
   181→            temperature=0.7,
   182→        )
   183→    elif mode == "inference":
   184→        # HF Inference Endpoints expose an OpenAI-compatible API
   185→        return OpenAiChatPromptDriver(
   186→            base_url="https://api-inference.huggingface.co/v1",
   187→            api_key=os.environ.get("HF_TOKEN", ""),
   188→            model=model,
   189→            max_tokens=1024,
   190→            temperature=0.7,
   191→        )
   192→    elif mode == "tgi":
   193→        if not hf_endpoint_url:
   194→            raise ValueError("TGI mode requires hf_endpoint_url parameter")
   195→        return OpenAiChatPromptDriver(
   196→            base_url=hf_endpoint_url,
   197→            api_key=os.environ.get("HF_TOKEN", ""),
   198→            model=model,
   199→            max_tokens=1024,
   200→            temperature=0.7,
   201→        )
   202→    else:
   203→        raise ValueError(f"Unknown HF inference mode: {mode}")

   310→def transform_query(tactical_issue: str, rules: dict | None = None) -> dict:
   311→    """
   312→    Strategic Query Transformation: translate a tactical issue into a
   313→    philosophical search string for GraphRAG retrieval.

   316→    in src/ava007/query_transform.ts. It enforces L6 authority guard —
   317→    only the Ava007 coordination layer may transform queries.
   318→
   319→    No LLM call. Zero tokens. Pure deterministic mapping.
   320→    """
   321→    active_rules = rules or DEFAULT_ABSTRACTION_RULES
   322→    issue_lower = tactical_issue.lower()
   323→
   324→    matched_abstraction = None
   325→    matched_keyword = None
   326→    for keyword, abstraction in active_rules.items():
   327→        if keyword in issue_lower:
   328→            matched_abstraction = abstraction
   329→            matched_keyword = keyword
   330→            break
   331→
   332→    if matched_abstraction is None:
   333→        matched_abstraction = "General_Inquiry"
   334→        matched_keyword = "none"
   335→
   336→    return {
   337→        "original_query": tactical_issue,
   338→        "abstraction": matched_abstraction,
   339→        "matched_keyword": matched_keyword,
   340→        "graphrag_search_string": f"{matched_abstraction}: {tactical_issue}",

   365→# 7. RUNTIME TIERS (mirrors src/ava007/coordination_types.ts)
   366→#    Reflex (~100 tokens, no LLM) -> Executive (Mellum2, ~500 tokens)
   367→#    -> Cortex (Mercury 2, 1k+ tokens)
   368→# =========================================================================
   369→
   370→def evaluate_reflex_gate(atom: dict) -> dict:
   371→    """
   372→    Reflex gate evaluation. Mirrors src/ava007/escalation_gates.ts
   373→    evaluateReflexGate(). Zero LLM call — pure rule matching.
   374→
   375→    Returns: { "target": "reflex"|"executive", "reason": str, "action": str|None }
   376→    """
   377→    atom_type = atom.get("type", "")
   378→    atom_source = atom.get("source", "")
   379→    confidence = atom.get("confidence", 0)
   380→    importance = atom.get("importance", "medium")
   381→    payload = atom.get("payload", {})
   382→    payload_bytes = len(json.dumps(payload).encode("utf-8"))
   383→
   384→    # Explicit escalation flags
   385→    if payload.get("requiresCortex"):
   386→        return {
   387→            "target": "executive",
   388→            "reason": "explicit_cortex_request",
   389→            "action": None,
   390→        }
   391→    if payload.get("requiresExecutive"):
   392→        return {
   393→            "target": "executive",
   394→            "reason": "explicit_executive_request",
   395→            "action": None,
   396→        }
   397→
   398→    # Document uploads always escalate
   399→    if atom_type == "document" or atom_source == "document_upload":
   400→        return {
   401→            "target": "executive",
   402→            "reason": "document_upload",
   403→            "action": None,
   404→        }
   405→
   406→    # Critical importance always escalates
   407→    if importance == "critical":
   408→        return {
   409→            "target": "executive",
   410→            "reason": "critical_importance",
   411→            "action": None,
   412→        }
   413→
   414→    # Low confidence escalates
   415→    if confidence < GATE_CONFIG["reflex_confidence_threshold"]:
   416→        return {
   417→            "target": "executive",
   418→            "reason": "ingestion_confidence_below_threshold",
   419→            "action": None,
   420→        }
   421→
   422→    # Known NFC shapes -> reflex resolution (0 LLM tokens)
   423→    if atom_source in GATE_CONFIG["reflex_nfc_sources"] or atom_type == "nfc_tap":
   424→        return {
   425→            "target": "reflex",
   426→            "reason": "known_nfc_shape",
   427→            "action": "resolve_nfc_tap",
   428→        }
   429→
   430→    # Known webhook shapes -> reflex resolution
   431→    if (
   432→        atom_source in GATE_CONFIG["reflex_webhook_sources"]
   433→        or atom_type in GATE_CONFIG["reflex_known_webhook_types"]
   434→    ):
   435→        return {
   436→            "target": "reflex",
   437→            "reason": "known_webhook_shape",
   438→            "action": "resolve_webhook",
   439→        }
   440→
   441→    # Payload over budget
   442→    if payload_bytes > GATE_CONFIG["reflex_max_payload_bytes"]:
   443→        return {
   444→            "target": "executive",
   445→            "reason": "payload_over_reflex_budget",
   446→            "action": None,
   447→        }
   448→
   449→    return {"target": "executive", "reason": "unknown_shape", "action": None}
   450→

   573→def process_atom(atom: dict, ava_agent: Agent, rag_engine: RagEngine) -> dict:
   574→    """
   575→    Full coordination loop:
   576→      Observe -> Interpret -> Orchestrate -> Verify -> Commit -> Anchor
   577→
   578→    Mirrors the TypeScript Ava007Orchestrator.processAtom() in
   579→    src/ava007/orchestrator.ts.
   580→
   581→    Tiers:
   582→      - Reflex: <5ms, no LLM, rule-based routing
   583→      - Executive: HuggingFace Gemma-2, ~500 tokens
   584→      - Cortex: HuggingFace Gemma-2, 1k+ tokens
   585→    """
   586→    started_at = datetime.now(timezone.utc)
   587→
   588→    # STEP 1: OBSERVE — Atom ingressed from NFC, A2A POST, webhook, etc.
   589→    atom_id = atom.get(
   590→        "id",
   591→        hashlib.sha256(json.dumps(atom).encode()).hexdigest()[:16],
   592→    )
   593→
   594→    # STEP 2: INTERPRET — CFGL rule routing determines the tier
   595→    reflex_gate = evaluate_reflex_gate(atom)
   596→
   597→    # STEP 3: ORCHESTRATE
   598→    if reflex_gate["target"] == "reflex" and reflex_gate["action"]:
   599→        # Reflex tier: No LLM call — zero token cost
   600→        return {
   601→            "atom_id": atom_id,
   602→            "tier": "reflex",
   603→            "action": reflex_gate["action"],
   604→            "gate_reason": reflex_gate["reason"],
   605→            "confidence": 1.0,
   606→            "latency_ms": _elapsed_ms(started_at),
   607→            "context_token_budget": GATE_CONFIG["reflex_context_token_budget"],
   608→        }
   609→
   610→    # Executive or Cortex tier: Use Griptape agent with RAG context
   611→    # Strategic query transformation — deterministic, zero LLM cost
   612→    atom_description = atom.get("payload", {}).get("description", str(atom))
   613→    transformation = transform_query(atom_description)
   614→    graphrag_search = transformation["graphrag_search_string"]
   615→
   616→    # RAG retrieval from Neo4j graph (depth<=5)
   617→    rag_context = rag_engine.process_query(graphrag_search)
   618→
   619→    # Build prompt for Griptape agent (executive/cortex)
   620→    orchestration_prompt = (
   621→        f"Context from graph retrieval: {rag_context}\n\n"
   622→        f"Atom to process: {json.dumps(atom)}\n\n"
   623→        f"Gate decision: {reflex_gate['target']} tier "
   624→        f"(reason: {reflex_gate['reason']})\n\n"
   625→        f"Strategic abstraction: {transformation['abstraction']}\n"
   626→        f"Execute the appropriate coordination action."
   627→    )
   628→
   629→    # Run agent — Griptape handles the executive/cortex HF inference call
   630→    result = ava_agent.run(orchestration_prompt)
   631→
   632→    # Determine tier based on gate
   633→    tier = reflex_gate["target"]
   634→    token_budget = (
   635→        GATE_CONFIG["executive_context_token_budget"]
   636→        if tier == "executive"
   637→        else GATE_CONFIG["cortex_context_token_budget"]
   638→    )
   639→
   640→    # STEP 4: VERIFY — Authority chain enforcement
   641→    # L6 sole decision authority confirmed by governance ruleset
   642→
   643→    # STEP 5: COMMIT — Write to memory (mirrors MemoryStore.append)
   644→    output_text = (
   645→        result.output_text if hasattr(result, "output_text") else str(result)
   646→    )
   647→
   648→    # STEP 6: ANCHOR — Tashi consensus finality
   649→    return {
   650→        "atom_id": atom_id,
   651→        "tier": tier,
   652→        "action": reflex_gate["action"] or "agent_orchestrated",
   653→        "gate_reason": reflex_gate["reason"],
   654→        "confidence": 0.9,
   655→        "latency_ms": _elapsed_ms(started_at),
   656→        "context_token_budget": token_budget,
   657→        "agent_output": output_text,
   658→        "strategic_abstraction": transformation["abstraction"],
   659→        "original_query": transformation["original_query"],
   660→    }

   663→def _elapsed_ms(started_at: datetime) -> float:
   664→    """Calculate elapsed milliseconds since start."""
   665→    delta = datetime.now(timezone.utc) - started_at
   666→    return delta.total_seconds() * 1000

   679→class AvaHandler(BaseHTTPRequestHandler):
   680→    """
   681→    HTTP handler that mirrors the legacy Ava.py POST API.
   682→    Receives a goal, runs it through the Griptape Agent,
   683→    and returns structured JSON output.
   684→
   685→    The agent runs a closed-loop execution:
   686→      Search -> Scrape -> Store -> RAG -> Reason -> Output
   687→
   688→    All tool outputs are stored off-prompt in TaskMemory.
   689→    The LLM sees metadata references only — never raw scraped data.
   690→    """
   691→
   692→    def do_POST(self):
   693→        try:
   694→            content_length = int(self.headers.get("Content-Length", 0))
   695→            post_data = self.rfile.read(content_length)
   696→            payload = json.loads(post_data.decode("utf-8"))
   697→
   698→            goal = payload.get("goal", "No goal provided")
   699→
   700→            # Determine if this is a structured atom or a simple goal
   701→            if "type" in payload and "source" in payload:
   702→                # Structured atom — run through coordination loop
   703→                result = process_atom(payload, _ava_agent, _rag_engine)
   704→                response_data = {
   705→                    "status": "coordination_complete",
   706→                    "goal": goal,
   707→                    "output": result,
   708→                }
   709→            else:
   710→                # Simple goal — run full Griptape Agent
   711→                result = _ava_agent.run(goal)
   712→                output_text = (
   713→                    result.output_text
   714→                    if hasattr(result, "output_text")
   715→                    else str(result)
   716→                )
   717→                response_data = {
   718→                    "status": "execution_complete",
   719→                    "goal": goal,
   720→                    "output": output_text,
   721→                }
   722→
   723→            self._send_json(200, response_data)
   724→
   725→        except json.JSONDecodeError:
   726→            self._send_json(400, {"error": "Invalid JSON payload"})
   727→        except Exception as e:
   728→            self._send_json(500, {"error": str(e)})
   729→
   730→    def _send_json(self, code: int, data: dict) -> None:
   731→        self.send_response(code)
   732→        self.send_header("Content-type", "application/json")
   733→        self.end_headers()
   734→        self.wfile.write(json.dumps(data, default=str).encode("utf-8"))
   735→
   736→    def log_message(self, format, *args):
   737→        # Clean logs — no per-request noise
   738→        return
